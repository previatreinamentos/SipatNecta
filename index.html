Conectar Planilha Google

Insira a URL do seu Google Apps Script (Web App) para armazenar em tempo real as respostas dos participantes na sua planilha.

  1. Crie uma nova Planilha no Google Drive.
  2. No menu superior, vá em Extensões > Apps Script.
  3. Apague o código padrão e cole o código abaixo:
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var doc = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = doc.getSheetByName("Respostas") || doc.getActiveSheet();
    
    // Cria cabeçalho se a planilha estiver vazia
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Data/Hora", "Nome", "Matrícula/RE", "Unidade", 
        "Acertos", "Total Questões", "Percentual (%)",
        "Q1", "Q2", "Q3", "Q4", "Q5", "Q6", 
        "Q7", "Q8", "Q9", "Q10", "Q11", "Q12"
      ]);
    }
    
    var data = JSON.parse(e.postData.contents);
    var row = [
      new Date().toLocaleString("pt-BR"),
      data.nome,
      data.matricula,
      data.unidade,
      data.acertos,
      data.total,
      data.percentual + "%"
    ];
    
    // Adiciona as 12 respostas
    for (var i = 1; i <= 12; i++) {
      row.push(data.respostas["q" + i] || "");
    }
    
    sheet.appendRow(row);
    return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", error: err.message})).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
  1. Clique em Implantar > Nova implantação.
  2. Tipo: selecione App da Web.
  3. Acesso: selecione Qualquer pessoa (necessário para que os colaboradores no celular possam enviar sem login no script).
  4. Clique em Implantar, autorize as permissões e copie a URL do App da Web no campo acima.
